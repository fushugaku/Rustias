cmake_minimum_required(VERSION 3.25)
project(RustiasPlugin VERSION 0.1.0 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJC OBJCXX)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0)
elseif(NOT WIN32)
  message(FATAL_ERROR "Supported plugin platforms are macOS and Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(RUSTIAS_PORTABLE_RUNTIME "Use QuickJS/Wasmtime (required on Windows)" ${WIN32})
set(RUSTIAS_RESOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../output/plugin/resources" CACHE PATH "Prepared shared web assets")
set(CLAP_WRAPPER_DOWNLOAD_DEPENDENCIES ON CACHE BOOL "Download pinned SDKs")
set(CLAP_WRAPPER_COPY_AFTER_BUILD OFF CACHE BOOL "Do not install automatically")
include(FetchContent)
FetchContent_Declare(clap_wrapper GIT_REPOSITORY https://github.com/free-audio/clap-wrapper.git GIT_TAG 1cca996e96f29ab2be7ae9f8cfe532bbc92e1dd6)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(clap_wrapper json)

if(RUSTIAS_PORTABLE_RUNTIME)
  set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(quickjs GIT_REPOSITORY https://github.com/quickjs-ng/quickjs.git GIT_TAG 6d46d07d04041b40f4f49eaa7fdebe44c314c699 EXCLUDE_FROM_ALL)
  FetchContent_MakeAvailable(quickjs)
  set(RUSTIAS_WASMTIME_ROOT "" CACHE PATH "Extracted Wasmtime 49.0.3 C API")
  if(NOT RUSTIAS_WASMTIME_ROOT)
    if(WIN32)
      set(wasm_asset x86_64-windows-c-api.zip)
      set(wasm_hash f8dabbc1df2358f15effbd6e50c940c03df3836e13cbf4f09cdf6369057aed01)
    elseif(APPLE AND CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
      set(wasm_asset aarch64-macos-c-api.tar.xz)
      set(wasm_hash 0c853eb1715b3730304f926d111d4b4a3605b961c5693753bfeeb6b19332c41b)
    else()
      message(FATAL_ERROR "Set RUSTIAS_WASMTIME_ROOT for this portable test architecture")
    endif()
    FetchContent_Declare(wasmtime URL "https://github.com/bytecodealliance/wasmtime/releases/download/v49.0.3/wasmtime-v49.0.3-${wasm_asset}" URL_HASH SHA256=${wasm_hash} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(wasmtime)
    set(RUSTIAS_WASMTIME_ROOT "${wasmtime_SOURCE_DIR}")
  endif()
  add_library(rustias_wasmtime STATIC IMPORTED)
  if(WIN32)
    set_target_properties(rustias_wasmtime PROPERTIES IMPORTED_LOCATION "${RUSTIAS_WASMTIME_ROOT}/lib/wasmtime.lib")
  else()
    set_target_properties(rustias_wasmtime PROPERTIES IMPORTED_LOCATION "${RUSTIAS_WASMTIME_ROOT}/lib/libwasmtime.a")
  endif()
  set_target_properties(rustias_wasmtime PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${RUSTIAS_WASMTIME_ROOT}/include")
  set(runtime_source src/RuntimePortable.cpp)
else()
  set(runtime_source src/Runtime.cpp)
endif()
add_library(rustias_runtime STATIC ${runtime_source})
target_include_directories(rustias_runtime PUBLIC src)
if(RUSTIAS_PORTABLE_RUNTIME)
  target_compile_definitions(rustias_runtime PUBLIC RUSTIAS_PORTABLE_RUNTIME LIBWASM_STATIC)
  target_link_libraries(rustias_runtime PUBLIC qjs rustias_wasmtime)
  if(WIN32)
    target_link_libraries(rustias_runtime PUBLIC ws2_32 bcrypt userenv advapi32 ntdll ole32 shell32)
  endif()
else()
  target_link_libraries(rustias_runtime PUBLIC "-framework JavaScriptCore")
endif()
if(WIN32)
  FetchContent_Declare(webview2 URL https://api.nuget.org/v3-flatcontainer/microsoft.web.webview2/1.0.3856.49/microsoft.web.webview2.1.0.3856.49.nupkg URL_HASH SHA256=bc0f76eb911b569838dc4aa8f8d325269b966bedb592863d26211aef3a099f1a DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(webview2)
  set(editor_source src/EditorWindows.cpp)
else()
  set(editor_source src/Editor.mm)
endif()
add_library(rustias_core OBJECT src/Plugin.cpp ${editor_source})
target_include_directories(rustias_core PUBLIC src)
target_link_libraries(rustias_core PUBLIC clap nlohmann_json::nlohmann_json rustias_runtime)
if(APPLE)
  target_compile_options(rustias_core PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>" -Wall -Wextra)
  target_link_libraries(rustias_core PUBLIC "-framework Cocoa" "-framework WebKit")
else()
  target_compile_definitions(rustias_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_include_directories(rustias_core PRIVATE "${webview2_SOURCE_DIR}/build/native/include")
  target_link_libraries(rustias_core PUBLIC "${webview2_SOURCE_DIR}/build/native/x64/WebView2LoaderStatic.lib" shlwapi version comdlg32)
endif()
add_library(rustias_clap MODULE)
target_link_libraries(rustias_clap PRIVATE rustias_core)
target_add_clap_configuration(TARGET rustias_clap OUTPUT_NAME Rustias BUNDLE_IDENTIFIER io.github.fushugaku.rustias.clap BUNDLE_VERSION 0.1.0 RESOURCE_DIRECTORY "${RUSTIAS_RESOURCES}")
set_property(TARGET rustias_clap APPEND PROPERTY LINK_DEPENDS "${RUSTIAS_RESOURCES}/runtime.js" "${RUSTIAS_RESOURCES}/web/host-app.js" "${RUSTIAS_RESOURCES}/rustias.wasm")
add_library(rustias_vst3 MODULE)
if(APPLE)
  add_library(rustias_au MODULE)
  target_add_auv2_wrapper(TARGET rustias_au OUTPUT_NAME Rustias BUNDLE_IDENTIFIER io.github.fushugaku.rustias.au BUNDLE_VERSION 0.1.0 MANUFACTURER_NAME Rustias MANUFACTURER_CODE Fshg SUBTYPE_CODE Rsts INSTRUMENT_TYPE aumu PREFER_CMAKE_AUV2_CONFIGURATION TRUE MACOS_EMBEDDED_CLAP_LOCATION "$<TARGET_BUNDLE_DIR:rustias_clap>")
  add_dependencies(rustias_au rustias_clap)
  # CMake's bundle plist generation otherwise overwrites the wrapper's AU registration.
  add_custom_command(TARGET rustias_au POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_BINARY_DIR}/rustias_au-build-helper-output/auv2_Info.plist" "$<TARGET_BUNDLE_DIR:rustias_au>/Contents/Info.plist")
  target_add_vst3_wrapper(TARGET rustias_vst3 OUTPUT_NAME Rustias BUNDLE_IDENTIFIER io.github.fushugaku.rustias.vst3 BUNDLE_VERSION 0.1.0 MACOS_EMBEDDED_CLAP_LOCATION "$<TARGET_BUNDLE_DIR:rustias_clap>")
  add_dependencies(rustias_vst3 rustias_clap)
  set_property(TARGET rustias_au APPEND PROPERTY LINK_DEPENDS "$<TARGET_FILE:rustias_clap>")
  set_property(TARGET rustias_vst3 APPEND PROPERTY LINK_DEPENDS "$<TARGET_FILE:rustias_clap>")
else()
  # Export the CLAP entry inside the VST3 module; the wrapper finds its own entry.
  target_link_libraries(rustias_vst3 PRIVATE rustias_core)
  target_add_vst3_wrapper(TARGET rustias_vst3 OUTPUT_NAME Rustias WINDOWS_FOLDER_VST3 TRUE)
  set_property(TARGET rustias_vst3 APPEND PROPERTY LINK_DEPENDS "${RUSTIAS_RESOURCES}/runtime.js" "${RUSTIAS_RESOURCES}/web/host-app.js")
  add_custom_command(TARGET rustias_vst3 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${RUSTIAS_RESOURCES}" "$<TARGET_FILE_DIR:rustias_vst3>/../Resources")
  add_custom_command(TARGET rustias_clap POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${RUSTIAS_RESOURCES}" "$<TARGET_FILE_DIR:rustias_clap>/Rustias.resources")
endif()
add_executable(rustias-runtime-check tests/runtime.cpp)
target_link_libraries(rustias-runtime-check PRIVATE rustias_runtime)
enable_testing()
add_test(NAME shared_worklet COMMAND rustias-runtime-check "${RUSTIAS_RESOURCES}")
add_executable(rustias-host-check tests/host.cpp)
target_link_libraries(rustias-host-check PRIVATE rustias_core)
add_test(NAME host_state COMMAND rustias-host-check "${RUSTIAS_RESOURCES}")
if(APPLE)
  add_executable(rustias-editor-check tests/editor.mm)
  target_compile_options(rustias-editor-check PRIVATE -fobjc-arc)
  add_executable(rustias-plugin-preview MACOSX_BUNDLE tests/preview.mm)
  target_link_libraries(rustias-plugin-preview PRIVATE rustias_core)
  target_compile_definitions(rustias-plugin-preview PRIVATE RUSTIAS_PREVIEW_RESOURCES="${RUSTIAS_RESOURCES}")
  target_compile_options(rustias-plugin-preview PRIVATE -fobjc-arc)
  set_target_properties(rustias-plugin-preview PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER io.github.fushugaku.rustias.preview)
else()
  add_executable(rustias-editor-check tests/editor-windows.cpp)
endif()
target_link_libraries(rustias-editor-check PRIVATE rustias_core)
add_test(NAME editor_workers COMMAND rustias-editor-check "${RUSTIAS_RESOURCES}")
set_tests_properties(editor_workers PROPERTIES TIMEOUT 45)

file(MAKE_DIRECTORY "${RUSTIAS_RESOURCES}/licenses")
function(rustias_license source name)
  if(EXISTS "${source}")
    configure_file("${source}" "${RUSTIAS_RESOURCES}/licenses/${name}.txt" COPYONLY)
  endif()
endfunction()
get_target_property(clap_headers clap INTERFACE_INCLUDE_DIRECTORIES)
list(GET clap_headers 0 clap_headers)
rustias_license("${clap_headers}/../LICENSE" CLAP)
rustias_license("${clap_wrapper_SOURCE_DIR}/LICENSE" clap-wrapper)
rustias_license("${json_SOURCE_DIR}/LICENSE.MIT" nlohmann-json)
rustias_license("${CMAKE_CURRENT_BINARY_DIR}/cpm/vst3sdk/LICENSE.txt" VST3-SDK)
rustias_license("${CMAKE_CURRENT_BINARY_DIR}/cpm/vst3sdk/base/LICENSE.txt" VST3-base)
rustias_license("${CMAKE_CURRENT_BINARY_DIR}/cpm/vst3sdk/pluginterfaces/LICENSE.txt" VST3-interfaces)
if(APPLE)
  rustias_license("${CMAKE_CURRENT_BINARY_DIR}/cpm/AudioUnitSDK/LICENSE.txt" AudioUnitSDK)
endif()
if(RUSTIAS_PORTABLE_RUNTIME)
  rustias_license("${quickjs_SOURCE_DIR}/LICENSE" QuickJS-NG)
  rustias_license("${RUSTIAS_WASMTIME_ROOT}/LICENSE" Wasmtime)
endif()
if(WIN32)
  rustias_license("${webview2_SOURCE_DIR}/LICENSE.txt" WebView2)
endif()
