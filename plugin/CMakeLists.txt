  set(native_library "${CMAKE_CURRENT_BINARY_DIR}/native/rustias_plugin_engine.lib")

  set(native_library "${CMAKE_CURRENT_BINARY_DIR}/native/librustias_plugin_engine.a")

  "${rustias_root}/crates/radias-synth-domain/src/*"
  "${rustias_root}/crates/radias-synth-application/src/*"
  "${rustias_root}/crates/radias-synth-infrastructure/src/*"
  "${rustias_root}/crates/radias-web/src/*"
  "${CMAKE_CURRENT_SOURCE_DIR}/engine/src/*")
add_custom_command(OUTPUT "${native_library}"
  COMMAND "${Python3_EXECUTABLE}" "${rustias_root}/scripts/build-plugin-engine.py"
    --output "${native_library}" --architectures "${CMAKE_OSX_ARCHITECTURES}"
  DEPENDS ${native_sources} "${rustias_root}/scripts/build-plugin-engine.py"
    engine/Cargo.toml engine/Cargo.lock
    "${rustias_root}/crates/radias-web/Cargo.toml"
    "${rustias_root}/crates/radias-synth-domain/Cargo.toml"
    "${rustias_root}/crates/radias-synth-application/Cargo.toml"
    "${rustias_root}/crates/radias-synth-infrastructure/Cargo.toml"
  VERBATIM)
add_custom_target(rustias_native_build DEPENDS "${native_library}")

function(rustias_license source name)
  if(EXISTS "${source}")
    configure_file("${source}" "${RUSTIAS_RESOURCES}/licenses/${name}.txt" COPYONLY)
  endif()
endfunction()
